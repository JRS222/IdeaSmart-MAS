Diagnosis

Three problems in order of severity.

Bug 1 — the FATAL. $rowParams = [ordered]@{} creates an OrderedDictionary. When PowerShell evaluates $rowParams[[int]$matches[1]], it has two indexer overloads to choose from — this[object key] and this[int index] — and it binds to the int indexer (positional access). On the very first row1text param, index 1 is out of bounds for a dictionary with 0 entries, so it throws ArgumentOutOfRangeException. That's the crash. Fix: use plain @{} (Hashtable), which has no ambiguity.

Bug 2 — silent data corruption waiting behind Bug 1. The actual header is:

```
Reg No | WO No | WC | Acronym | CC | Equip No | Stts | Crew | Dt Issued | Dt Due | WO Description
```

11 columns, none named Class, Equip, Desc, Priority, or Due. The current fallback indices return the wrong column for each of those fields — you'd silently get Equip mapped to CC, Priority mapped to Crew, etc. Needs fixed header-name matching.

Bug 3 — 144,205 params = ~144,000 work orders. The empty pRegNo= isn't filtering by register, so the server dumps every open WO at the site. Fetching per-WO detail for 144K rows would take hours. Need to pre-filter in the parse loop by Dt Issued or Dt Due (both available without detail fetch) so only the ~few hundred in-window rows survive to the detail stage.

Also worth noting: Dt Closed and Hours To Date are not in the search response. FilterBy = 'Closed' can't work from search data alone — it requires a detail fetch per row, which is exactly what Bug 3 makes infeasible. I'm changing the default to Due.

```powershell
function ConvertFrom-WoSearchResponse {
    param(
        [string]$Content,
        [string]$SiteId,
        [string]$Tag = 'search',
        [string]$FilterMode = '',       # 'Due' | 'Issued' | '' (no pre-filter)
        [DateTime]$WindowFrom = [DateTime]::MinValue,
        [DateTime]$WindowTo   = [DateTime]::MinValue
    )

    if ([string]::IsNullOrWhiteSpace($Content)) {
        Write-Log "ConvertFrom-WoSearchResponse[$Tag]: empty response"
        return ,@()
    }
    $Content = $Content.Trim()
    Write-Log "ConvertFrom-WoSearchResponse[$Tag]: $($Content.Length) bytes"

    # ---------- XML ----------
    if ($Content -match '^\s*<\?xml' -or $Content -match '^\s*<\s*ROWSET') {
        Write-Log "ConvertFrom-WoSearchResponse[$Tag]: XML branch"
        $xml = New-Object System.Xml.XmlDocument
        $xml.LoadXml($Content)
        $out = @()
        foreach ($row in @($xml.SelectNodes('//ROW'))) {
            $fields = @{}
            foreach ($child in $row.ChildNodes) {
                if ($child.NodeType -eq [System.Xml.XmlNodeType]::Element) {
                    $fields[$child.Name.ToUpper()] = "$($child.InnerText)".Trim()
                }
            }
            $out += [PSCustomObject]@{
                WoPtr       = $fields['WO_PTR']
                WorkOrderNo = $fields['WONO']
                RegisterNo  = $fields['REGNO']
                WorkCode    = $fields['WORK_CODE']
                Acronym     = $fields['ACRONYM']
                ClassCode   = $fields['CLASSCODE']
                Equipment   = $fields['EQUIP_NO']
                Status      = $fields['WO_STATUS']
                Crew        = $fields['CREW_ID']
                DateIssued  = $fields['DATE_ISSUED']
                DueDate     = $fields['DATE_DUE']
                Description = $fields['WO_DESC']
                SiteId      = $SiteId
                RawCells    = @()
            }
        }
        return ,$out
    }

    $sw = [System.Diagnostics.Stopwatch]::StartNew()

    # ---------- APPLET <param> format ----------
    $paramMatches = [regex]::Matches(
        $Content,
        '(?is)<param\b[^>]*?\bname\s*=\s*["''](?<name>[^"'']+)["''][^>]*?\bvalue\s*=\s*["''](?<val>[^"'']*)["'']'
    )
    Write-Log "ConvertFrom-WoSearchResponse[$Tag]: $($paramMatches.Count) <param> element(s) in $([int]$sw.Elapsed.TotalMilliseconds)ms"

    if ($paramMatches.Count -gt 0) {
        # PLAIN HASHTABLE, not [ordered]@{} — OrderedDictionary's int
        # indexer fights the key lookup and throws ArgumentOutOfRange.
        $rowParams   = @{}
        $headers     = @()
        $otherParams = New-Object System.Collections.Generic.List[string]

        foreach ($m in $paramMatches) {
            $pname = $m.Groups['name'].Value.Trim()
            $pval  = [System.Net.WebUtility]::HtmlDecode($m.Groups['val'].Value)
            if ($pname -match '^row(\d+)text$') {
                $rowParams[[int]$matches[1]] = $pval
            } elseif ($pname -ieq 'colheadertext') {
                $headers = @($pval -split ',' | ForEach-Object { $_.Trim() })
                Write-Log "ConvertFrom-WoSearchResponse[$Tag]: columns = $($headers -join ' | ')"
            } elseif ($pname -notmatch '^row(\d+)') {
                # Log every non-row param name once, so we can see
                # whether the WO pointer is delivered as its own param
                # (e.g. "rowptrs=123,456,789" or "ptr1text=...").
                if (-not $otherParams.Contains($pname)) {
                    $otherParams.Add($pname) | Out-Null
                }
            }
        }
        Write-Log "ConvertFrom-WoSearchResponse[$Tag]: $($rowParams.Count) data row(s), $($headers.Count) column(s)"
        Write-Log "ConvertFrom-WoSearchResponse[$Tag]: non-row params = $($otherParams -join ', ')"

        if ($rowParams.Count -gt 0) {
            # Positional fallbacks derived from the real header
            # (Reg No | WO No | WC | Acronym | CC | Equip No | Stts | Crew | Dt Issued | Dt Due | WO Description)
            $idx = @{
                RegNo       = 0
                WoNo        = 1
                WorkCode    = 2
                Acronym     = 3
                ClassCode   = 4
                EquipNo     = 5
                Status      = 6
                Crew        = 7
                DateIssued  = 8
                DueDate     = 9
                Description = 10
            }

            $out         = @()
            $preFiltered = 0
            $processed   = 0

            foreach ($rowIdx in ($rowParams.Keys | Sort-Object { [int]$_ })) {
                $processed++
                if (($processed % 10000) -eq 0) {
                    Write-Log "ConvertFrom-WoSearchResponse[$Tag]: processed $processed / $($rowParams.Count) row(s), kept so far: $($out.Count)"
                }

                $raw = "$($rowParams[$rowIdx])"
                # Pick a delimiter.  Try each candidate; prefer the one
                # whose split length is at least 11 (our known column count).
                $cells = $null
                foreach ($sep in @('|', ';', '~', "`t")) {
                    $parts = @($raw -split [regex]::Escape($sep))
                    if ($parts.Count -ge 11) { $cells = $parts; break }
                }
                if (-not $cells) {
                    # Last resort: comma, then keep whole string
                    $parts = @($raw -split ',')
                    if ($parts.Count -ge 11) { $cells = $parts } else { $cells = @($raw) }
                }
                $cells = @($cells | ForEach-Object { "$_".Trim() })

                # Quick date pre-filter (columns 8/9 come free with the row)
                if ($FilterMode -in @('Issued', 'Due') -and
                    $WindowFrom -ne [DateTime]::MinValue -and
                    $WindowTo   -ne [DateTime]::MinValue) {
                    $colIdx = if ($FilterMode -eq 'Issued') { $idx.DateIssued } else { $idx.DueDate }
                    if ($colIdx -lt $cells.Count) {
                        $dt = $null
                        try { $dt = [DateTime]$cells[$colIdx] } catch { }
                        if (-not $dt -or $dt.Date -lt $WindowFrom.Date -or $dt.Date -gt $WindowTo.Date) {
                            $preFiltered++
                            continue
                        }
                    } else {
                        $preFiltered++
                        continue
                    }
                }

                $lookup = @{}
                for ($i = 0; $i -lt $cells.Count; $i++) {
                    if ($i -lt $headers.Count -and $headers[$i]) {
                        $lookup[$headers[$i]] = $cells[$i]
                    }
                }

                $out += [PSCustomObject]@{
                    WoPtr       = if ($cells.Count -gt 11) { $cells[11] } else { '' }
                    RegisterNo  = Get-ParamField $lookup @('Reg No')       $idx.RegNo       $cells
                    WorkOrderNo = Get-ParamField $lookup @('WO No')        $idx.WoNo        $cells
                    WorkCode    = Get-ParamField $lookup @('WC')           $idx.WorkCode    $cells
                    Acronym     = Get-ParamField $lookup @('Acronym')      $idx.Acronym     $cells
                    ClassCode   = Get-ParamField $lookup @('CC')           $idx.ClassCode   $cells
                    Equipment   = Get-ParamField $lookup @('Equip No')     $idx.EquipNo     $cells
                    Status      = Get-ParamField $lookup @('Stts')         $idx.Status      $cells
                    Crew        = Get-ParamField $lookup @('Crew')         $idx.Crew        $cells
                    DateIssued  = Get-ParamField $lookup @('Dt Issued')    $idx.DateIssued  $cells
                    DueDate     = Get-ParamField $lookup @('Dt Due')       $idx.DueDate     $cells
                    Description = Get-ParamField $lookup @('WO Description') $idx.Description $cells
                    SiteId      = $SiteId
                    RawCells    = $cells
                }
            }

            Write-Log "ConvertFrom-WoSearchResponse[$Tag]: kept $($out.Count) row(s) after pre-filter (dropped $preFiltered)"
            return ,$out
        }
    }

    # ---------- Fallback: regex <tr> parse ----------
    $trMatches = [regex]::Matches(
        $Content,
        '(?is)<tr\b[^>]*\bclass\s*=\s*["'']?(?:MAIN|HILITE)["'']?[^>]*>(?<body>.*?)</tr>'
    )
    Write-Log "ConvertFrom-WoSearchResponse[$Tag]: $($trMatches.Count) <tr class=MAIN|HILITE> in $([int]$sw.Elapsed.TotalMilliseconds)ms"
    if ($trMatches.Count -eq 0) {
        $trMatches = [regex]::Matches($Content, '(?is)<tr\b[^>]*>(?<body>.*?)</tr>')
        Write-Log "ConvertFrom-WoSearchResponse[$Tag]: broad <tr> count = $($trMatches.Count)"
    }

    $out = @()
    foreach ($tr in $trMatches) {
        $cellMatches = [regex]::Matches($tr.Groups['body'].Value, '(?is)<td\b[^>]*>(?<c>.*?)</td>')
        if ($cellMatches.Count -lt 5) { continue }
        $cells = @($cellMatches | ForEach-Object {
            $t = $_.Groups['c'].Value -replace '(?is)<[^>]+>', ' '
            $t = [System.Net.WebUtility]::HtmlDecode($t)
            ($t -replace '\s+', ' ').Trim()
        })
        $woIdx = -1
        for ($i = 0; $i -lt [Math]::Min($cells.Count, 4); $i++) {
            if ($cells[$i] -match '^\d{3,}$') { $woIdx = $i; break }
        }
        if ($woIdx -lt 0) { continue }
        $out += [PSCustomObject]@{
            WoPtr       = if ($cells.Count -gt 11) { $cells[11] } else { '' }
            RegisterNo  = if ($woIdx-1 -ge 0) { $cells[$woIdx-1] } else { '' }
            WorkOrderNo = $cells[$woIdx]
            WorkCode    = if ($woIdx+1 -lt $cells.Count) { $cells[$woIdx+1] } else { '' }
            Acronym     = if ($woIdx+2 -lt $cells.Count) { $cells[$woIdx+2] } else { '' }
            ClassCode   = if ($woIdx+3 -lt $cells.Count) { $cells[$woIdx+3] } else { '' }
            Equipment   = if ($woIdx+4 -lt $cells.Count) { $cells[$woIdx+4] } else { '' }
            Status      = if ($woIdx+5 -lt $cells.Count) { $cells[$woIdx+5] } else { '' }
            Crew        = if ($woIdx+6 -lt $cells.Count) { $cells[$woIdx+6] } else { '' }
            DateIssued  = if ($woIdx+7 -lt $cells.Count) { $cells[$woIdx+7] } else { '' }
            DueDate     = if ($woIdx+8 -lt $cells.Count) { $cells[$woIdx+8] } else { '' }
            Description = if ($woIdx+9 -lt $cells.Count) { $cells[$woIdx+9] } else { '' }
            SiteId      = $SiteId
            RawCells    = $cells
        }
    }

    if ($out.Count -eq 0) {
        Save-RawResponse -Content $Content -Tag "WoSearch_$Tag" | Out-Null
        Write-Log "ConvertFrom-WoSearchResponse[$Tag]: ZERO rows -- inspect the Debug Dumps file"
    } else {
        Write-Log "ConvertFrom-WoSearchResponse[$Tag]: parsed $($out.Count) row(s) in $([int]$sw.Elapsed.TotalMilliseconds)ms"
    }
    return ,$out
}
```

```powershell
function Get-EmarsWorkOrdersAll {
    param(
        [string]$SiteId,
        [string[]]$StatusCodes,
        [string]$SingleRegister = '',
        [int]$MaxRegs = 30,
        [int]$MaxRows = 5000,
        [string]$FilterMode = '',
        [DateTime]$WindowFrom = [DateTime]::MinValue,
        [DateTime]$WindowTo   = [DateTime]::MinValue
    )

    $seen = @{}
    $out  = @()

    if ([string]::IsNullOrWhiteSpace($SingleRegister)) {
        foreach ($st in $StatusCodes) {
            $q = "pSite=$SiteId&pWOStatus=$st&pRegNo=&pWONo="
            try {
                $body = Invoke-EmarsGet -Path 'em_work_pkg.em_getWOData' -Query $q -TimeoutSec 30
            } catch {
                Write-Log "Get-EmarsWorkOrdersAll: empty-reg status '$st' FAILED -- $($_.Exception.Message)"
                continue
            }
            $wos = ConvertFrom-WoSearchResponse -Content $body -SiteId $SiteId -Tag "empty_${st}" `
                    -FilterMode $FilterMode -WindowFrom $WindowFrom -WindowTo $WindowTo
            Write-Log "Get-EmarsWorkOrdersAll: empty-reg status '$st' -> $(@($wos).Count) WO(s) after pre-filter"

            foreach ($wo in @($wos)) {
                if ([string]::IsNullOrWhiteSpace($wo.WorkOrderNo)) { continue }
                if ($seen.ContainsKey($wo.WorkOrderNo)) { continue }
                $seen[$wo.WorkOrderNo] = $true
                $out += $wo
                if ($MaxRows -gt 0 -and $out.Count -ge $MaxRows) {
                    Write-Log "Get-EmarsWorkOrdersAll: MaxRows ($MaxRows) reached -- stopping"
                    return ,$out
                }
            }
        }
        if ($out.Count -gt 0) { return ,$out }
        Write-Log "Get-EmarsWorkOrdersAll: empty-reg path returned 0 -- iterating registers"
    }

    $regs = @()
    if (-not [string]::IsNullOrWhiteSpace($SingleRegister)) {
        $regs = @($SingleRegister)
    } else {
        $regs = Get-EmarsRegisterList -SiteId $SiteId
        if ($regs.Count -eq 0) {
            $regs = $script:FallbackRegisters
            Write-Log "Get-EmarsWorkOrdersAll: no register list -- using fallback list of $($regs.Count)"
        }
        if ($MaxRegs -gt 0 -and $regs.Count -gt $MaxRegs) {
            Write-Log "Get-EmarsWorkOrdersAll: capping at $MaxRegs register(s)"
            $regs = @($regs | Select-Object -First $MaxRegs)
        }
    }

    foreach ($reg in $regs) {
        foreach ($st in $StatusCodes) {
            $q = "pSite=$SiteId&pWOStatus=$st&pRegNo=$reg&pWONo="
            try {
                $body = Invoke-EmarsGet -Path 'em_work_pkg.em_getWOData' -Query $q -TimeoutSec 25
            } catch {
                Write-Log "  reg=$reg status=$st FAILED -- $($_.Exception.Message)"
                continue
            }
            $wos = ConvertFrom-WoSearchResponse -Content $body -SiteId $SiteId -Tag "reg${reg}_${st}" `
                    -FilterMode $FilterMode -WindowFrom $WindowFrom -WindowTo $WindowTo
            Write-Log "  reg=$reg status=$st -> $(@($wos).Count) WO(s) after pre-filter"
            foreach ($wo in @($wos)) {
                if ([string]::IsNullOrWhiteSpace($wo.WorkOrderNo)) { continue }
                if ($seen.ContainsKey($wo.WorkOrderNo)) { continue }
                $seen[$wo.WorkOrderNo] = $true
                $out += $wo
                if ($MaxRows -gt 0 -and $out.Count -ge $MaxRows) {
                    Write-Log "Get-EmarsWorkOrdersAll: MaxRows ($MaxRows) reached -- stopping"
                    return ,$out
                }
            }
        }
    }
    return ,$out
}
```
