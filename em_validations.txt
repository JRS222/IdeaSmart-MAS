function chkdate(field) {
// dd mon yy
 var ddmonyyyy=/(^\d{1,2})\s(\D{3})\s(\d{2}$|\d{4}$)/;
// mm-dd-yyyy and validates leap year
 var mmddyyyy=/^(?:(?:(?:0?[13578]|1[02])(\/|-)31)|(?:(?:0?[1,3-9]|1[0-2])(\/|-)(?:29|30)))(\/|-)(?:[1-9]\d\d\d|\d[1-9]\d\d|\d\d[1-9]\d|\d\d\d[1-9])$|^(?:(?:0?[1-9]|1[0-2])(\/|-)(?:0?[1-9]|1\d|2[0-8]))(\/|-)(?:[1-9]\d\d\d|\d[1-9]\d\d|\d\d[1-9]\d|\d\d\d[1-9])$|^(0?2(\/|-)29)(\/|-)(?:(?:0[48]00|[13579][26]00|[2468][048]00)|(?:\d\d)?(?:0[48]|[2468][048]|[13579][26]))$/;
// ddmmyyyy, no separator
 var alldigits=/(^\d{2})(\d{2})(\d{2}$|\d{4}$)/;
 var mmddyy=/(^\d{1,2})\D(\d{1,2})\D(\d{2})$/;
// simple mm/dd/yyyy to force "/" as separator
 var dateSlashes=/(^\d{1})\D(\d{1,2})\D(\d{4})/;
 var monAR = new Array("Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec");
 var moStr='';
 var moNum=0;
 var yrStr='';
 
 var fieldStr=field.value;
 
  if (fieldStr.match(ddmonyyyy)) {
     fieldStr=fieldStr.toLowerCase();
     moStr=fieldStr.replace(ddmonyyyy,"$2");
     yrStr=fieldStr.replace(ddmonyyyy,"$3");
     for (d=0; d<monAR.length; d++) {
        if (moStr==monAR[d].toLowerCase()) {
           moNum=d+1;
        }
     }
    if (yrStr.length==2) {
       yrStr='20'+yrStr;
    }
    fieldStr=fieldStr.replace(ddmonyyyy,moNum+"/"+"$1/"+yrStr);
  }
  if (fieldStr.match(alldigits)) {
    var dNow = new Date();
    var numYear = dNow.getYear();
    yrStr=fieldStr.replace(alldigits,"$3");
    if (yrStr.length==2) {
       yrStr='20'+yrStr;
    }
    fieldStr=fieldStr.replace(alldigits,"$1/$2/"+yrStr);
  } 
  if (fieldStr.match(mmddyy)) {
    yrStr=fieldStr.replace(mmddyy,"$3");
    yrStr='20'+yrStr;
    fieldStr=fieldStr.replace(mmddyy,"$1/$2/"+yrStr);
  }
  if (fieldStr.match(dateSlashes)) {
    fieldStr=fieldStr.replace(dateSlashes,"0$1/0$2/$3");
  }
  if (fieldStr.match(mmddyyyy)) {
    fieldStr=fieldStr.replace(/\D/g,"/");
    field.value=fieldStr;
    try {
       createDateTitle(field,fieldStr);
    } catch(e) { field.title='error doing title'; }
    return 0;
  }
   return 1;
}

   var finalDate=/(^\d{1,2})\D(\d{1,2})\D(\d{4}$)/;

   var d=fieldStr.replace(finalDate,"$2");
   var y=fieldStr.replace(finalDate,"$3");
   var monthAR = new Array("none","January","February","March","April","May","June","July","August","September","October","November","December");
   var dateString='';
   dateString=monthAR[m]+' '+d+', '+y;
   field.title=dateString;   
 
}

function checkNSNformat(field) {
   //match for any non-digit (anything that's not 0-9)
   var nonNum = /\D+/;
   //a pattern for matching a 13 character long pattern -- digits only.
   var pattern = /(\d{4})(\d{2})(\d{3})(\d{4})/g;
   
   //var nsnPattern=/(\d{4,4})(-(\d{2})-(\d{3})-(\d{4})){0,1}/g;
   
   // matches patterns in either of the following formats
   // 1) XXXX (where X is any valid digit 0-9)
   // 2) XXXX-XX-XXX-XXXX (where X is any valid digit 0-9)
   //var nsnPattern=/(\d{4})(-(\d{2})-(\d{3})-(\d{4})){0,1}/g;
   
   var nsnPattern1 = /(\d{4})/;
   var nsnPattern2 = /(\d{4})-(\d{2})-(\d{3})-(\d{4})/;
   var nsnPattern3 = /(\d{4})-(\d{2})/;
   
   //fieldStr will be whatever value was typed in
   var fieldStr = field.value;
   var newStr = "";
   
   if (fieldStr.length > 0) {
      if (fieldStr.match(nsnPattern1) && fieldStr.length == 4) {
          return true;
      } else if (fieldStr.match(nsnPattern2) && fieldStr.length == 16) {
          return true;
      } else if (fieldStr.match(nsnPattern3) && fieldStr.length == 7) {
          return true;
      } else {
         //split the passed in string at all the non-digit positions, if any exist.
         pieces = fieldStr.split(nonNum);
         fieldStr = "";
         
         //now put fieldStr back together as a concatenation of all the parts (basically removing any non-digit characters).
         for (i=0; i<pieces.length; i++) {
            fieldStr = fieldStr+pieces[i];
         }

         //now if fieldStr matches the pattern format above (13 char digit-only string)
         if (fieldStr.match(pattern)) {
            newStr = fieldStr.replace(pattern,"$1-$2-$3-$4");
            //02-03-2005 cprice added second = sign to make the correct comparison
            //if the length is 16, then replace the field on screen with the new (formatted) value
            if (newStr.length == 16) {
               field.value = newStr;
            } else {
               return false;
            }
         } else {
            return false;
         }
      }
   }
   
   return true;
}

function isValidTime(fieldName) {
   var field=d[fieldName];

   var isValid=true;

      var timeAR=val.split(":");
      if (isNumeric(timeAR[0]) && isNumeric(timeAR[1])) {
         var hr=parseInt(timeAR[0]);
         var mi=parseInt(timeAR[1]);

            isValid=false;
         }
      } else {
         isValid=false;
